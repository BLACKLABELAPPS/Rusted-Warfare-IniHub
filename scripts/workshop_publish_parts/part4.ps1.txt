      throw "STEAM_USERDATA_ID is required when using STEAM_LOCALCONFIG_VDF."
    }
    if ($safeUserDataId -notmatch "^\d+$") {
      throw "Invalid STEAM_USERDATA_ID. Expected numeric SteamID3 folder."
    }
    $localConfigDir = Join-Path (Join-Path (Join-Path $SteamRoot "userdata") $safeUserDataId) "config"
    if (-not (Test-Path -LiteralPath $localConfigDir)) {
      New-Item -ItemType Directory -Path $localConfigDir -Force | Out-Null
    }
    $localConfigBytes = [Convert]::FromBase64String($LocalConfigBase64)
    [IO.File]::WriteAllBytes((Join-Path $localConfigDir "localconfig.vdf"), $localConfigBytes)
    $restoredAny = $true
    Write-Step "Restored userdata/$safeUserDataId/config/localconfig.vdf session file."
  }

  if (-not $restoredAny) {
    throw "No session file provided. Set STEAM_CONFIG_VDF or STEAM_LOCALCONFIG_VDF."
  }

  $safeSsfnName = [IO.Path]::GetFileName([string]$SsfnName)
  $safeSsfnBase64 = [string]$SsfnBase64
  if ([string]::IsNullOrWhiteSpace($safeSsfnName) -or [string]::IsNullOrWhiteSpace($safeSsfnBase64)) {
    Write-Step "SSFN file not provided. Continuing with session file(s) only."
    return
  }
  if (-not $safeSsfnName.ToLowerInvariant().StartsWith("ssfn")) {
    throw "Invalid STEAM_SSFN_FILE_NAME value"
  }

  $ssfnBytes = [Convert]::FromBase64String($safeSsfnBase64)
  [IO.File]::WriteAllBytes((Join-Path $SteamRoot $safeSsfnName), $ssfnBytes)
}
